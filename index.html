#include <WiFi.h>
#include <WebServer.h>
#include <ESP32Servo.h>

// --- UBAH DENGAN WIFI RUMAH / HOTSPOT HP KAMU ---
const char* ssid = "DESKTOP11";
const char* password = "11111111";

WebServer server(80);

// Pin Hardware
#define SENSOR_PROXIMITY 18
#define LED_CONVEYOR     25

#define SERVO_BASE_PIN     17  // Servo 1
#define SERVO_ELBOW_PIN    16  // Servo 2
#define SERVO_SHOULDER_PIN 26  // Servo 3
#define SERVO_GRIPPER_PIN  33  // Servo 4

// Objek Servo
Servo servoBase;      // Servo 1
Servo servoElbow;     // Servo 2
Servo servoShoulder;  // Servo 3
Servo servoGripper;   // Servo 4

// Variabel Status Hardware
bool conveyorStatus = true;
bool proximityStatus = false;
bool isSortingProcess = false; 

// Posisi Awal Sesuai Setup Contoh Kode Kamu
int baseAngle = 0;       // Servo 1 = 0
int elbowAngle = 30;     // Servo 2 = 30
int shoulderAngle = 180; // Servo 3 = 180
int gripperAngle = 20;   // Servo 4 = 20

// Web UI dengan Library CDN Online
const char HTML_PAGE[] PROGMEM = R"rawliteral(

void handleRoot() {
  server.send(200, "text/html", HTML_PAGE);
}

void handleStatus() {
  String json = "{";
  json += "\"conveyor\":" + String(conveyorStatus ? "true" : "false") + ",";
  json += "\"proximity\":" + String(proximityStatus ? "true" : "false") + ",";
  json += "\"base\":" + String(baseAngle) + ",";
  json += "\"shoulder\":" + String(shoulderAngle) + ",";
  json += "\"elbow\":" + String(elbowAngle) + ",";
  json += "\"gripper\":" + String(gripperAngle);
  json += "}";

  server.send(200, "application/json", json);
}

// --- FUNGSI PENGGERAK SERVO TUNGGAL SECARA HALUS (SMOOTH WRITE) ---
void smoothWrite(Servo &s, int ¤tAngle, int targetAngle, int speedDelay = 15) {
  int step = (currentAngle < targetAngle) ? 1 : -1;
  while (currentAngle != targetAngle) {
    currentAngle += step;
    s.write(currentAngle);
    
    // Tetap layani request Web UI agar simulasi 3D tidak freeze
    server.handleClient();
    delay(speedDelay);
  }
}

// --- URUTAN GERAKAN PEMILAHAN LOGAM SESUAI LOGIKA KAMU ---
void runSortingSequence() {
  isSortingProcess = true;

  // 1. Matikan Conveyor
  conveyorStatus = false;
  digitalWrite(LED_CONVEYOR, LOW);
  delay(500);

  // Servo 1 gerak ke 90
  smoothWrite(servoBase, baseAngle, 90);
  delay(800);

  // Servo 4 gerak ke 90
  smoothWrite(servoGripper, gripperAngle, 90);
  delay(800);

  // Servo 2 gerak ke 90
  smoothWrite(servoElbow, elbowAngle, 90);
  delay(800);

  // Servo 3 gerak ke 50
  smoothWrite(servoShoulder, shoulderAngle, 50);
  delay(800);

  // Servo 4 gerak ke 115
  smoothWrite(servoGripper, gripperAngle, 115);
  delay(800);

  // Servo 3 gerak ke 90
  smoothWrite(servoShoulder, shoulderAngle, 90);
  delay(800);

  // Servo 1 gerak ke 0 (Kembali / Putar)
  smoothWrite(servoBase, baseAngle, 0);
  delay(800);

  // Servo 3 gerak ke 50
  smoothWrite(servoShoulder, shoulderAngle, 50);
  delay(800);

  // Servo 4 gerak ke 90
  smoothWrite(servoGripper, gripperAngle, 90);
  delay(800);

  // Servo 3 gerak ke 90
  smoothWrite(servoShoulder, shoulderAngle, 90);
  delay(800);

  // Servo 2 gerak ke 30
  smoothWrite(servoElbow, elbowAngle, 30);
  delay(800);

  // 2. Jalankan Kembali Conveyor
  conveyorStatus = true;
  digitalWrite(LED_CONVEYOR, HIGH);

  isSortingProcess = false;
}

void setup() {
  Serial.begin(115200);

  pinMode(SENSOR_PROXIMITY, INPUT_PULLUP);
  pinMode(LED_CONVEYOR, OUTPUT);
  digitalWrite(LED_CONVEYOR, HIGH);

  ESP32PWM::allocateTimer(0);
  ESP32PWM::allocateTimer(1);
  ESP32PWM::allocateTimer(2);
  ESP32PWM::allocateTimer(3);

  servoBase.setPeriodHertz(50);
  servoElbow.setPeriodHertz(50);
  servoShoulder.setPeriodHertz(50);
  servoGripper.setPeriodHertz(50);

  servoBase.attach(SERVO_BASE_PIN, 500, 2400);
  servoElbow.attach(SERVO_ELBOW_PIN, 500, 2400);
  servoShoulder.attach(SERVO_SHOULDER_PIN, 500, 2400);
  servoGripper.attach(SERVO_GRIPPER_PIN, 500, 2400);

  // Set Posisi Awal Sesuai Setup Contoh Kamu
  servoBase.write(baseAngle);         // 0
  servoElbow.write(elbowAngle);       // 30
  servoShoulder.write(shoulderAngle); // 180
  servoGripper.write(gripperAngle);   // 20

  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  
  Serial.println("\nWiFi Connected!");
  Serial.print("Buka IP ini di Browser: ");
  Serial.println(WiFi.localIP());

  server.on("/", handleRoot);
  server.on("/status", handleStatus);
  server.begin();
  Serial.println("HTTP Server Started");
}

void loop() {
  server.handleClient();

  if (!isSortingProcess) {
    bool isMetalDetected = (digitalRead(SENSOR_PROXIMITY) == LOW);
    proximityStatus = isMetalDetected;

    if (isMetalDetected) {
      runSortingSequence();
    }
  }
}
